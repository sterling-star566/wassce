name: Build Android APK
on:
  push:
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Find project folder
        run: |
          P=$(find . -name package.json -not -path "*/node_modules/*" | head -1)
          [ -z "$P" ] && echo "package.json not found - upload all project files" && exit 1
          echo "APP_DIR=$(dirname "$P")" >> $GITHUB_ENV
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17
      - name: Install and add Android
        working-directory: ${{ env.APP_DIR }}
        run: |
          npm install
          npx cap add android
          M=android/app/src/main/AndroidManifest.xml
          sed -i 's#</manifest>#<uses-permission android:name="android.permission.POST_NOTIFICATIONS"/><uses-permission android:name="android.permission.SCHEDULE_EXACT_ALARM"/><uses-permission android:name="android.permission.USE_EXACT_ALARM"/><uses-permission android:name="android.permission.RECEIVE_BOOT_COMPLETED"/></manifest>#' $M
          cp -r resources/res/. android/app/src/main/res/
          npx cap sync android
      - name: Build debug APK
        working-directory: ${{ env.APP_DIR }}/android
        run: |
          chmod +x gradlew
          ./gradlew assembleDebug --no-daemon
      - uses: actions/upload-artifact@v4
        with:
          name: wassce-tracker-apk
          path: ${{ env.APP_DIR }}/android/app/build/outputs/apk/debug/app-debug.apk
          if-no-files-found: error
